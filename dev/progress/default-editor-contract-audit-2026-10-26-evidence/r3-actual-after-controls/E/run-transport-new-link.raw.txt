from pathlib import Path
import json, subprocess, time
base=Path(__file__).resolve().parent;episode=base/'transport-episode-2';session='D-E-r3-controlled-session';a='D-E-R3-transport-old-failure';b='D-E-R3-transport-old-busy';records=[]
import sys

from gates import verify
runtime_gate=verify('E',native=False,require_launch=True)
def control(value):
    with (episode/'fake-control.jsonl').open('a') as f:f.write(json.dumps(value)+'\n')
    time.sleep(.2)
def held(id):return [v for v in map(json.loads,(episode/'fake-io.jsonl').read_text().splitlines()) if v.get('type')=='response-held' and v.get('id')==id]
def step(n,stage,**action):
    label=f'{n:02}-R3-transport-{stage}';action['label']=label
    r=subprocess.run(['rtk','proxy','node',str(base/'driver.cjs'),json.dumps(action)],capture_output=True,text=True)
    with (base/'driver-stdout.jsonl').open('a') as f:f.write(json.dumps({'action':action,'exit':r.returncode,'stdout':r.stdout,'stderr':r.stderr})+'\n')
    records.append(label)
    if r.returncode:raise RuntimeError(r.stdout+r.stderr)
    value=json.loads((base/'steps'/(label+'.json')).read_text());print(json.dumps({'label':label,'result':value.get('result')}),flush=True);return value
